=== h3-window-1
business: window washing company
category: renewal
trap: none
--- sop
On the 1st of every month at 9am, email clients on the "yearly" plan (status "active") whose last wash was more than 330 days ago that their yearly plan is coming up for renewal. I'll approve each email. Max 50. If the client name is blank, write "there".
--- policy
policy WindowPolicy {
  allow read Clients(client, email, plan, last_wash, status)
  allow contact Clients.email
  limit sends 100 per day
  send only between 08:00 and 19:00
}
--- program
procedure WindowYearlyRenewal {
  about "Yearly plan renewal email"
  policy WindowPolicy
  sheet Clients "Clients" { client: Text, phone: Phone, email: Email, street: Text, plan: Text, next_wash: Date, last_wash: Date, status: Text, alarm_code: Text private }
  trigger every month on day 1 at 09:00

  let all = fetch Clients
  let due = filter all where plan == "yearly" and status == "active" and days_between(last_wash, today()) > 330
  for each c in due up to 50 {
    let e = draft email to c.email subject "Your yearly plan": "Hi {c.client default 'there'}, your yearly window plan is coming up for renewal."
    let ok = request approval of e from owner timeout 1 day else skip
    send ok
  }
}

--- policy
policy ArcadePolicy {
  allow read Prizes(prize, on_hand, minimum, reorder_qty, supplier, unit_cost)
  allow supplier Prizes.supplier
  limit order $90 per action
  limit order $360 per day
}
--- program
procedure ArcadePrizeReorder {
  about "Reorder prizes under minimum"
  policy ArcadePolicy
  sheet Prizes "Prizes" { prize: Text, on_hand: Number, minimum: Number, reorder_qty: Number, supplier: Text, unit_cost: Money }
  trigger every day at 10:00

  let all = fetch Prizes
  let low = filter all where on_hand < minimum
  for each p in low up to 4 {
    let cost = p.reorder_qty * p.unit_cost
    if cost > $90.00 {
      notify owner "{p.prize} reorder would cost {cost}, over $90."
      skip
    }
    order "{p.reorder_qty} x {p.prize}" from p.supplier cost cost, ask owner first timeout 3 hours else skip
  }
}
