=== h3-violin-1
business: violin repair shop
category: quote-followup
trap: none
--- sop
Every day at 10am: repairs with status "quoted" that were quoted more than 5 days ago and still aren't approved by the customer. Email them a polite follow-up about the estimate. I approve each. Max 15. If the customer name is empty, write "there".
--- policy
policy ViolinPolicy {
  allow read Repairs(customer, email, instrument, quoted_on, approved, status)
  allow contact Repairs.email
  limit sends 40 per day
  send only between 09:00 and 18:00
}
--- program
procedure ViolinQuoteFollowUp {
  about "Follow up on unanswered estimates"
  policy ViolinPolicy
  sheet Repairs "Repairs" { customer: Text, phone: Phone, email: Email, instrument: Text, quoted_on: Date, estimate: Money, approved: Bool, status: Text }
  trigger every day at 10:00

  let all = fetch Repairs
  let waiting = filter all where status == "quoted" and approved == false and days_between(quoted_on, today()) > 5
  for each r in waiting up to 15 {
    let e = draft email to r.email subject "Your repair estimate": "Hi {r.customer default 'there'}, just checking in about the estimate for your {r.instrument}."
    let ok = request approval of e from owner timeout 1 day else skip
    send ok
  }
}
