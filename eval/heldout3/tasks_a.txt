  if c.deposit > $120.00 {
    notify owner "Cancellation {c.rental_id}: deposit {c.deposit} is over $120."
    stop "over refund limit"

  if c.deposit <= $30.00 {
    refund c.deposit for c.rental_id reason "rain cancellation"
  } else {
    refund c.deposit for c.rental_id reason "rain cancellation", ask owner first timeout 1 day else stop "refund not approved"
  }
}

  }
