=== thrift-01
business: thrift store
category: receipts
trap: none
--- sop
Every evening at 7: donations where we haven't sent a receipt yet get an emailed thank-you with the estimated value for their taxes. Approve as one batch. Then mark receipt_sent as yes. Up to 50. Blank cells: the message uses the donor, value; if the donor is blank, write "there" instead; if the value is blank, write "not assessed" instead.
--- policy
policy ThriftPolicy {
  allow read Donations(donor, email, value, receipt_sent)
  allow write Donations(receipt_sent)
  allow contact Donations.email
  limit sends 100 per day
  send only between 08:00 and 20:00
  approval batch valid 1 hour max 50 preview 5
}
--- program
procedure ThriftReceipts {
  about "Donation thank-you receipts"
  policy ThriftPolicy
  sheet Donations "Donations" { donor: Text, email: Email, dropped_on: Date, value: Money, receipt_sent: Bool, status: Text }
  trigger every day at 19:00

  let all = fetch Donations
  let todo = filter all where receipt_sent == false
  let msgs = draft each d in todo up to 50 email to d.email subject "Thank you for your donation": "Hi {d.donor default 'there'}, thank you! Estimated value for your records: {d.value default 'not assessed'}."
  let ok = request batch approval of msgs from owner timeout 1 hour else stop "batch not approved"
  send ok
  for each d in todo up to 50 {
    update d set receipt_sent = true
  }
}

=== it-04
business: computer repair shop
category: reorder
trap: none
--- sop
Daily 8am: parts below minimum get the reorder quantity ordered from their supplier, with my OK. Never more than $200 an order — tell me instead if so. At most 3 parts a day.
--- policy
policy RepairPolicy {
  allow read Parts(part, qty, minimum, reorder_qty, supplier, unit_cost)
  allow supplier Parts.supplier
  limit order $200 per action
  limit order $600 per day
}
--- program
procedure RepairPartsReorder {
  about "Reorder low parts"
  policy RepairPolicy
  sheet Parts "Parts" { part: Text, qty: Number, minimum: Number, reorder_qty: Number, supplier: Text, unit_cost: Money }
  trigger every day at 08:00

  let all = fetch Parts
  let low = filter all where qty < minimum
  for each p in low up to 3 {
    let cost = p.reorder_qty * p.unit_cost
    if cost > $200.00 {
      notify owner "{p.part} reorder would cost {cost}, over $200."
      skip
    }
    order "{p.reorder_qty} x {p.part}" from p.supplier cost cost, ask owner first timeout 4 hours else skip
  }
}

=== it-05
business: computer repair shop
category: wait-followup
trap: none
--- sop
When a new ticket is opened: email the customer an acknowledgement with their ticket number (I approve). Wait 3 days. If the ticket is still "new" by then, tell me it hasn't been touched. Blank cells: the message uses the customer, ticket; if the customer is blank, write "there" instead; if the ticket is blank, don't send that message.
--- policy
policy RepairPolicy {
  allow read Tickets(ticket, customer, email, status)
  allow contact Tickets.email
  limit sends 80 per day
  send only between 09:00 and 19:00
}
--- program
procedure RepairAckAndWatch {
  about "Acknowledge a ticket and flag it if untouched"
  policy RepairPolicy
  sheet Tickets "Tickets" { ticket: Text, customer: Text, email: Email, opened: Date, device: Text, status: Text, password: Text private }
  trigger when new row t in Tickets

  let e = draft email to t.email subject "We got your device": "Hi {t.customer default 'there'}, your ticket number is {t.ticket}."
  let ok = request approval of e from owner timeout 1 day else stop "acknowledgement not approved"
  send ok
  wait 3 days
  let all = fetch Tickets
  let now_t = find all where ticket == t.ticket else stop "ticket removed"
  if now_t.status == "new" {
    notify owner "Ticket {now_t.ticket} has not been touched in 3 days."
  }
}
