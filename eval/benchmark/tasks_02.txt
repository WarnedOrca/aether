=== escape-01
business: escape room
category: reminder
trap: none
--- sop
9am every day: groups booked for today (status "booked") get a text telling the leader to arrive 15 minutes early. I approve each text. There are never more than 30. Blank cells: the message uses the group name, slot; if the group name is blank, write "team" instead; if the slot is blank, don't send that message.
--- policy
policy EscapePolicy {
  allow read Bookings(group_name, leader_phone, slot, status)
  allow contact Bookings.leader_phone
  deny reveal Bookings.waiver_notes
  limit sends 80 per day
  send only between 09:00 and 21:00
}
--- program
procedure EscapeSameDay {
  about "Arrive-early reminder on the day"
  policy EscapePolicy
  sheet Bookings "Bookings" { group_name: Text, leader_phone: Phone, slot: DateTime, players: Number, status: Text, waiver_notes: Text private }
  trigger every day at 09:00

  let all = fetch Bookings
  let today_b = filter all where date_of(slot) == today() and status == "booked"
  for each b in today_b up to 30 {
    let m = draft sms to b.leader_phone: "Hi {b.group_name default 'team'}! Your escape room is at {time_of(b.slot)} today. Please arrive 15 minutes early."
    let ok = request approval of m from owner timeout 3 hours else skip
    send ok
  }
}

--- sop
New booking row: if it's for fewer than 2 players, tell me and stop.   limit sends 80 per day
  send only between 09:00 and 21:00
}
--- program
procedure EscapeNewBooking {
  about "Confirm and schedule a new booking"
  policy EscapePolicy
  sheet Bookings "Bookings" { group_name: Text, leader_phone: Phone, slot: DateTime, players: Number, status: Text, waiver_notes: Text private }
  calendar Rooms "Rooms"
  trigger when new row b in Bookings

  if b.players < 2 {
    notify owner "Booking {b.group_name} is for fewer than 2 players."
    stop "too few players"

  let ev = book Rooms { title: "{b.group_name}", at: b.slot, length: 60 minutes }

Daily at 6am: supplies below minimum get reordered (the reorder quantity, from the listed supplier). Always ask me. No single order over $80; if it'd be more, just tell me. At most 3 items a day.
--- policy
policy LaundryPolicy {
  allow read Supplies(item, on_hand, minimum, reorder_qty, supplier, unit_cost)
  allow supplier Supplies.supplier
  limit order $80 per action
  limit order $240 per day
}
--- program
procedure LaundryReorder {
  about "Reorder supplies below minimum"
  policy LaundryPolicy
  sheet Supplies "Supplies" { item: Text, on_hand: Number, minimum: Number, reorder_qty: Number, supplier: Text, unit_cost: Money }
  trigger every day at 06:00

  let all = fetch Supplies
  let low = filter all where on_hand < minimum
  for each s in low up to 3 {
    let cost = s.reorder_qty * s.unit_cost
    if cost > $80.00 {
      notify owner "{s.item} reorder would cost {cost}, over $80."
      skip
    }
    order "{s.reorder_qty} x {s.item}" from s.supplier cost cost, ask owner first timeout 4 hours else skip
  }
}

    stop "over order limit"
