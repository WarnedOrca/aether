=== wash-01
business: car wash
category: renewal-reminder
trap: none
--- sop
Daily at 9: active members whose membership renews in 3 days get an email heads-up. I approve each. Up to 50. Blank cells: the message uses the member, plan, renews; if the member is blank, write "there" instead; if the plan is blank, just leave it out; if the renews is blank, don't send that message.

  allow read Memberships(member, email, plan, renews, status)

  deny reveal Memberships.card_last4
  limit sends 120 per day
  send only between 08:00 and 20:00
}
--- program
procedure WashRenewal {
  about "Renewal heads-up three days ahead"
  policy WashPolicy
  sheet Memberships "Memberships" { member: Text, email: Email, plan: Text, renews: Date, status: Text, card_last4: Text private }
  trigger every day at 09:00

  let all = fetch Memberships
  let soon = filter all where renews == today() + 3 days and status == "active"
  for each m in soon up to 50 {
    let e = draft email to m.email subject "Your membership renews soon": "Hi {m.member default 'there'}, your {m.plan default ''} plan renews on {m.renews}."
    let ok = request approval of e from owner timeout 1 day else skip
    send ok
  }
}

=== optom-04
business: optometrist
category: waitlist
trap: none
--- sop
When an exam is cancelled (new row in Cancellations): remove it from the Exams calendar using the event id, then find whoever joined the Waitlist earliest and is still "waiting", text them that a spot opened (I approve), and mark them "offered". If nobody's waiting, stop. Blank cells: the message uses the patient, slot text; if the patient is blank, write "there" instead; if the slot text is blank, write "call us for the time" instead.
--- policy
policy OptomPolicy {
  allow read Cancellations(event_id, slot_text)
  allow read Waitlist(patient, phone, joined, status)
  allow write Waitlist(status)
  allow contact Waitlist.phone
  allow calendar Exams
  limit sends 80 per day
  send only between 08:00 and 19:00
}
--- program
procedure OptomFillSlot {
  about "Offer a cancelled exam slot to the waitlist"
  policy OptomPolicy
  sheet Cancellations "Cancellations" { event_id: Text, slot_text: Text }
  sheet Waitlist "Waitlist" { patient: Text, phone: Phone, joined: Date, status: Text }
  calendar Exams "Exams"
  trigger when new row c in Cancellations

  cancel Exams event c.event_id
  let wl = fetch Waitlist
  let waiting = filter wl where status == "waiting"
  let first_one = sort waiting by joined first 1
  let p = find first_one where status == "waiting" else stop "nobody waiting"
  let m = draft sms to p.phone: "Hi {p.patient default 'there'}, an exam slot opened: {c.slot_text default 'call us for the time'}. Reply to take it."
  let ok = request approval of m from owner timeout 2 hours else stop "offer not approved"
  send ok
  update p set status = "offered"
}

  }

Blank cells: the message uses the host, deposit; if the host is blank, write "there" instead; if the deposit is blank, don't send that message.
--- policy
policy IcePolicy {
  allow read PartyCancellations(ref, host, phone, deposit, party_on)
  allow contact PartyCancellations.phone
  limit refund $100 per action
  limit refund $300 per day
  refund without approval up to $40
  limit sends 40 per day
  send only between 09:00 and 20:00
}
--- program
procedure IcePartyRefund {
  about "Refund a party deposit within the rules"
  policy IcePolicy
  sheet PartyCancellations "PartyCancellations" { ref: Text, host: Text, phone: Phone, deposit: Money, party_on: Date }
  trigger when new row c in PartyCancellations

  if days_between(today(), c.party_on) < 3 {
    notify owner "Cancellation {c.ref} is less than 3 days out: no refund."
    stop "too late for refund"
  }
  if c.deposit > $100.00 {
    notify owner "Cancellation {c.ref} deposit {c.deposit} is over $100."
    stop "over refund limit"
  }
  if c.deposit <= $40.00 {
    refund c.deposit for c.ref reason "party cancelled"
  } else {
    refund c.deposit for c.ref reason "party cancelled", ask owner first timeout 1 day else stop "refund not approved"
  }
  let m = draft sms to c.phone: "Hi {c.host default 'there'}, your deposit of {c.deposit} is on its way back."
  let ok = request approval of m from owner timeout 1 day else stop "message not approved"
  send ok
}

Monday at 8am: for "booked" parties in the next 7 days, how many are there and what's the biggest number of kids at any one? Tell me.
--- policy
policy IcePolicy {
  allow read Parties(party_on, kids, status)
}
--- program
procedure IceWeekParties {
  about "This week's parties"
  policy IcePolicy
  sheet Parties "Parties" { host: Text, phone: Phone, party_on: Date, kids: Number, deposit: Money, status: Text }
  trigger every monday at 08:00

  let all = fetch Parties
  let week = filter all where status == "booked" and days_between(today(), party_on) <= 7 and days_between(today(), party_on) >= 0
  let n = summarize count of week
  let biggest = summarize max kids of week
  notify owner "{n} parties this week; biggest has {biggest default '0'} kids."
}
