=== physio-01
business: physiotherapy clinic
category: reminder
trap: none
--- sop
Hey, so every morning at 8 I want our patients reminded about sessions they have in two days. Only ones still marked "booked". Texts are fine. I'll look at each one before it goes. Once it's sent, flip the status to "reminded". We never have more than 40 in a day. Blank cells: the message uses the patient, session at; if the patient is blank, write "there" instead; if the session at is blank, don't send that message.
--- policy
policy PhysioPolicy {
  allow read Sessions(patient, mobile, session_at, status)
  allow write Sessions(status)
  allow contact Sessions.mobile
  deny reveal Sessions.injury
  limit sends 100 per day
  send only between 07:30 and 20:00
}
--- program
procedure PhysioTwoDayReminder {
  about "Remind patients two days before their session"
  policy PhysioPolicy
  sheet Sessions "Sessions" { patient: Text, mobile: Phone, session_at: DateTime, therapist: Text, status: Text, injury: Text private }
  trigger every day at 08:00

  let all = fetch Sessions
  let soon = filter all where date_of(session_at) == today() + 2 days and status == "booked"
  for each s in soon up to 40 {
    let m = draft sms to s.mobile: "Hi {s.patient default 'there'}, see you in two days at {time_of(s.session_at)}."
    let ok = request approval of m from owner timeout 12 hours else skip
    send ok
    update s set status = "reminded"
  }
}

=== hw-03
business: hardware store
category: refund
trap: none
--- sop
New return in the Returns sheet: if the item was opened, tell me and stop — no refunds on opened stuff. If it was bought more than 30 days ago, also tell me and stop. If it's over $100, I deal with it myself. Otherwise refund it, but every refund needs my OK. After that, text the customer (I approve) and mark the return "refunded". Blank cells: the message uses the customer, amount; if the customer is blank, write "there" instead; if the amount is blank, don't send that message.
--- policy
policy HardwarePolicy {
  allow read Returns(return_no, customer, phone, amount, bought, opened)
  allow write Returns(status)
  allow contact Returns.phone
  limit refund $100 per action
  limit refund $400 per day
  limit sends 80 per day
  send only between 08:00 and 19:00
}
--- program
procedure HardwareReturn {
  about "Process returns within the rules"
  policy HardwarePolicy
  sheet Returns "Returns" { return_no: Text, customer: Text, phone: Phone, amount: Money, bought: Date, opened: Bool, status: Text }
  trigger when new row r in Returns

  if r.opened == true {
    notify owner "Return {r.return_no} is an opened item."
    stop "opened items not refundable"
  }
  if days_between(r.bought, today()) > 30 {
    notify owner "Return {r.return_no} is past 30 days."
    stop "outside return window"
  }
  if r.amount > $100.00 {
    notify owner "Return {r.return_no} is {r.amount}, over $100."
    stop "over refund limit"
  }
  refund r.amount for r.return_no reason "store return", ask owner first timeout 1 day else stop "refund not approved"
  update r set status = "refunded"
  let m = draft sms to r.phone: "Hi {r.customer default 'there'}, your refund of {r.amount} is on its way."
  let ok = request approval of m from owner timeout 1 day else stop "message not approved"
  send ok
}

  policy TattooPolicy

  trigger every day at 14:00

  let all = fetch Appointments
  let healing = filter all where status == "done" and date_of(start) == today() - 3 days
  for each a in healing up to 20 {
    let m = draft sms to a.phone: "Hi {a.client default 'there'}, how is your new tattoo healing? Reply with any questions."

=== tattoo-05
business: tattoo studio
category: new-row-branch
trap: none
--- sop
When a new appointment is added: if the deposit is under $50, text them asking for the deposit (I approve) and stop there. If it's $50 or more, put it on the Chairs calendar for 2 hours with the client's name, and text a confirmation (I approve). Blank cells: the message uses the client, start; if the client is blank, write "there" instead; if the start is blank, don't send that message.
--- policy
policy TattooPolicy {
  allow read Appointments(client, phone, start, deposit)
  allow contact Appointments.phone
  allow calendar Chairs
  limit sends 50 per day
  send only between 10:00 and 20:00
}
--- program
procedure TattooNewBooking {
  about "Deposit check, then book a chair"
  policy TattooPolicy
  sheet Appointments "Appointments" { client: Text, phone: Phone, start: DateTime, artist: Text, deposit: Money, status: Text, design_notes: Text private }
  calendar Chairs "Chairs"
  trigger when new row a in Appointments

  if a.deposit < $50.00 {
    let ask = draft sms to a.phone: "Hi {a.client default 'there'}, please pay the $50 deposit to hold your appointment."
    let ok1 = request approval of ask from owner timeout 6 hours else stop "deposit request not approved"
    send ok1
    stop "waiting for deposit"
  }
  let ev = book Chairs { title: "{a.client}", at: a.start, length: 2 hours }
  let m = draft sms to a.phone: "Hi {a.client default 'there'}, you're booked for {a.start}. See you then!"
  let ok = request approval of m from owner timeout 6 hours else stop "confirmation not approved"
  send ok
}
