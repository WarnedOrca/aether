--- policy
policy BrewPolicy {
  allow read Kegs(beer, on_hand, reorder_count, partner, keg_price)
  allow supplier Kegs.partner
  limit order $250 per action
  limit order $750 per day
}
--- program
procedure BrewKegReorder {
  about "Reorder low kegs"
  policy BrewPolicy
  sheet Kegs "Kegs" { beer: Text, on_hand: Number, reorder_count: Number, partner: Text, keg_price: Money }
  trigger every day at 09:00

  let all = fetch Kegs
  let low = filter all where on_hand < 2
  for each k in low up to 3 {
    let cost = k.reorder_count * k.keg_price
    if cost > $250.00 {
      notify owner "{k.beer} reorder would cost {cost}, over $250."
      skip
    }
    order "{k.reorder_count} kegs of {k.beer}" from k.partner cost cost, ask owner first timeout 3 hours else skip
  }
}

=== h2-camp-2
business: campground
category: refund
trap: none
--- sop
New row in Cancellations: if arrival is less than 5 days away, no refund, tell me and stop. If the deposit is over $150, I'll handle it. Deposits of $50 or less refund automatically; bigger ones need my OK.
--- policy
policy CampgroundPolicy {
  allow read Cancellations(booking, deposit, arrival)
  limit refund $150 per action
  limit refund $600 per day
  refund without approval up to $50
}
--- program
procedure CampCancelRefund {
  about "Deposit refund for cancellations"
  policy CampgroundPolicy
  sheet Cancellations "Cancellations" { booking: Text, camper: Text, deposit: Money, arrival: Date }
  trigger when new row c in Cancellations

  if days_between(today(), c.arrival) < 5 {
    notify owner "Cancellation {c.booking} is under 5 days out: no refund."
    stop "too late"
  }
  if c.deposit > $150.00 {
    notify owner "Cancellation {c.booking} deposit {c.deposit} is over $150."
    stop "over refund limit"
  }
  if c.deposit <= $50.00 {
    refund c.deposit for c.booking reason "cancelled stay"
  } else {
    refund c.deposit for c.booking reason "cancelled stay", ask owner first timeout 1 day else stop "refund not approved"
  }
}
