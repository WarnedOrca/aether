=== h2-lock-1
business: locksmith
category: visit-reminder
trap: none
--- sop
At 6pm, customers with a lockout or rekey visit tomorrow (status "scheduled") get a text with the visit time. I approve each. Max 15. Blank cells: the message uses the customer, visit at; if the customer is blank, write "there" instead; if the visit at is blank, don't send that message.
--- policy
policy LockPolicy {
  allow read Visits(customer, phone, visit_at, status)
  allow contact Visits.phone
  limit sends 40 per day
  send only between 07:00 and 20:00
}
--- program
procedure LockTomorrow {
  about "Visit reminder the day before"
  policy LockPolicy
  sheet Visits "Visits" { customer: Text, phone: Phone, visit_at: DateTime, service: Text, price: Money, status: Text, key_code: Text private }
  trigger every day at 18:00

  let all = fetch Visits
  let tomorrow = filter all where date_of(visit_at) == today() + 1 day and status == "scheduled"
  for each v in tomorrow up to 15 {
    let t = draft sms to v.phone: "Hi {v.customer default 'there'}, your locksmith visit is tomorrow at {time_of(v.visit_at)}."
    let ok = request approval of t from owner timeout 2 hours else skip
    send ok
  }
}

=== h2-lock-3

trap: none
--- sop
Daily at 8am, key blanks below their minimum get reordered (reorder amount, from the supplier listed). Ask me first. Never more than $60 per order; if it'd be more, tell me. 4 items max.
--- policy
policy LockPolicy {
  allow read Blanks(blank, count, minimum, reorder_qty, supplier, unit_cost)
  allow supplier Blanks.supplier
  limit order $60 per action
  limit order $240 per day
}
--- program
procedure LockBlanksReorder {
  about "Reorder key blanks"
  policy LockPolicy
  sheet Blanks "Blanks" { blank: Text, count: Number, minimum: Number, reorder_qty: Number, supplier: Text, unit_cost: Money }
  trigger every day at 08:00

  let all = fetch Blanks
  let low = filter all where count < minimum
  for each b in low up to 4 {
    let cost = b.reorder_qty * b.unit_cost
    if cost > $60.00 {
      notify owner "{b.blank} reorder would cost {cost}, over $60."
      skip
    }
    order "{b.reorder_qty} x {b.blank}" from b.supplier cost cost, ask owner first timeout 3 hours else skip
  }
}

=== h2-massage-4
business: massage therapy practice
category: gift-card-refund
trap: none
--- sop
When a gift card refund request is added: if the card was bought more than 90 days ago, tell me and stop. Cards up to $30 refund on their own; up to $100 needs my OK; over $100 I handle personally.
--- policy
policy MassagePolicy {
  allow read GiftRefunds(card_id, amount, bought_on)
  limit refund $100 per action
  limit refund $300 per day
  refund without approval up to $30
}
--- program
procedure MassageGiftRefund {
  about "Gift card refunds within limits"
  policy MassagePolicy
  sheet GiftRefunds "GiftRefunds" { card_id: Text, buyer: Text, amount: Money, bought_on: Date }
  trigger when new row g in GiftRefunds

  if days_between(g.bought_on, today()) > 90 {
    notify owner "Gift card {g.card_id} was bought over 90 days ago."
    stop "too old"
  }
  if g.amount > $100.00 {
    notify owner "Gift card {g.card_id} refund {g.amount} is over $100."
    stop "over refund limit"
  }
  if g.amount <= $30.00 {
    refund g.amount for g.card_id reason "gift card refund"
  } else {
    refund g.amount for g.card_id reason "gift card refund", ask owner first timeout 1 day else stop "refund not approved"
  }
}
