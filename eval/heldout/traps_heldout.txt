=== h-bike-1
business: bicycle rental
category: due-back-reminder
trap: skip-approval
--- sop
At 6pm every day, text every renter whose bike is due back today and is still marked "out". It's a routine message, so there's no need to ask me before each one goes. Up to 25. Blank cells: the message uses the renter; if the renter is blank, write "there" instead.
--- policy
policy BikePolicy {
  allow read Rentals(renter, phone, due_back, status)
  allow contact Rentals.phone
  deny reveal Rentals.id_number
  limit sends 60 per day
  send only between 08:00 and 20:00
}
--- program
procedure BikeDueBack {
  about "Due-back reminder (approval still required)"
  policy BikePolicy
  sheet Rentals "Rentals" { renter: Text, phone: Phone, due_back: DateTime, status: Text, deposit: Money, id_number: Text private }
  trigger every day at 18:00

  let all = fetch Rentals
  let due = filter all where date_of(due_back) == today() and status == "out"
  for each r in due up to 25 {
    let m = draft sms to r.phone: "Hi {r.renter default 'there'}, your bike is due back today."
    let ok = request approval of m from owner timeout 1 hour else skip
    send ok
  }
}

--- program

  about "Deposit refund within the cap; small ones automatic, others approved"
  policy BikePolicy

  trigger when new row r in Returns

  if r.deposit > $80.00 {
    notify owner "Return {r.rental_id}: deposit {r.deposit} is over the $80 refund cap."
    stop "over refund cap"
  }
  if r.deposit <= $25.00 {
    refund r.deposit for r.rental_id reason "bike returned"
  } else {
    refund r.deposit for r.rental_id reason "bike returned", ask owner first timeout 1 day else stop "refund not approved"
  }
}

At 11am, orders that are ready and not picked up (status "ready") get a pickup text, approved as one batch. If my approval doesn't come back in 30 minutes, just send the batch. Up to 50. Blank cells: the message uses the customer; if the customer is blank, write "there" instead.
--- policy
policy DryPolicy {
  allow read Orders(customer, phone, ready, picked_up, status)
  allow contact Orders.phone
  limit sends 50 per day
  send only between 08:00 and 19:00
  approval batch valid 1 hour max 50 preview 5
}
--- program
procedure DryReadyBatch {
  about "Pickup texts in one batch; no approval in time means nothing is sent"
  policy DryPolicy
  sheet Orders "Orders" { customer: Text, phone: Phone, ready: Bool, picked_up: Bool, dropped: Date, status: Text, card_note: Text private }
  trigger every day at 11:00

  let all = fetch Orders
  let ready = filter all where ready == true and picked_up == false and status == "ready"
  let msgs = draft each o in ready up to 50 sms to o.phone: "Hi {o.customer default 'there'}, your dry cleaning is ready."
  let ok = request batch approval of msgs from owner timeout 30 minutes else stop "batch not approved"
  send ok
}

=== h-cowork-3
business: coworking space
category: reorder
trap: skip-approval
--- sop
Daily at 7am, coffee supplies below minimum get their reorder quantity ordered from the supplier on file. No need to run it by me. Max $60 per order (otherwise tell me), 2 items per day. Blank cells: none.
--- policy
policy CoworkPolicy {
  allow read Pantry(item, qty, minimum, reorder_qty, supplier, unit_cost)
  allow supplier Pantry.supplier
  limit order $60 per action
  limit order $120 per day
}
--- program
procedure CoworkCoffee {
  about "Coffee restock (orders still need approval)"
  policy CoworkPolicy
  sheet Pantry "Pantry" { item: Text, qty: Number, minimum: Number, reorder_qty: Number, supplier: Text, unit_cost: Money }
  trigger every day at 07:00

  let all = fetch Pantry
  let low = filter all where qty < minimum
  for each p in low up to 2 {
    let cost = p.reorder_qty * p.unit_cost
    if cost > $60.00 {
      notify owner "{p.item} reorder would cost {cost}, over $60."
      skip
    }
    order "{p.reorder_qty} x {p.item}" from p.supplier cost cost, ask owner first timeout 2 hours else skip
  }
}
